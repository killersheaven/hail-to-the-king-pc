cmake_minimum_required(VERSION 3.16)
project(HailToTheKing C)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
# JIT=ON: the one-click build. No game code is compiled in; at launch the program
# compiles the C translated from the player's disc with the built-in TinyCC
# (runtime/jit.c). JIT=OFF (build.bat / build.sh): the translated C in build/gen
# is compiled ahead of time with the rest.
option(JIT "compile the game code at launch with the built-in TinyCC" OFF)
if(JIT)
  set(INTERP_DIR ${CMAKE_SOURCE_DIR}/build/interp CACHE PATH "folder with interp_ops.c (recomp.py --interp-only)")
  # embed runtime/sh4ctx.h: the generated code is compiled against this exact copy
  file(READ ${CMAKE_SOURCE_DIR}/runtime/sh4ctx.h SH4CTX_HEX HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," SH4CTX_BYTES "${SH4CTX_HEX}")
  file(WRITE ${CMAKE_BINARY_DIR}/sh4ctx_h.c "const char sh4ctx_h_text[] = {${SH4CTX_BYTES}0};\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/runtime/sh4ctx.h)
  set(GEN_SRC runtime/jit.c ${CMAKE_BINARY_DIR}/sh4ctx_h.c ${INTERP_DIR}/interp_ops.c third_party/tinycc/libtcc.c)
  set_source_files_properties(third_party/tinycc/libtcc.c PROPERTIES COMPILE_OPTIONS "-w")
else()
  file(GLOB GEN_SRC ${CMAKE_SOURCE_DIR}/build/gen/*.c)
  list(APPEND GEN_SRC runtime/code_aot.c)
endif()
add_executable(hail-to-the-king runtime/runtime.c runtime/hw.c runtime/bios_hle.c runtime/disc.c runtime/pvr.c runtime/dma.c runtime/maple.c runtime/aica.c runtime/aica_synth.c runtime/arm7.c runtime/coro.c runtime/interp.c runtime/render_gl.c runtime/upscale.c runtime/png.c runtime/textpatch.c runtime/iso9660.c runtime/sfd.c runtime/models.c runtime/hdfont.c ${GEN_SRC})
if(JIT)
  target_include_directories(hail-to-the-king PRIVATE runtime third_party/tinycc)
  target_compile_definitions(hail-to-the-king PRIVATE DC_JIT=1)
  if(NOT WIN32)
    target_link_libraries(hail-to-the-king PRIVATE dl pthread)
  endif()
else()
  target_include_directories(hail-to-the-king PRIVATE runtime build/gen)
endif()
# the SH-4 code relies on wrapping 32-bit arithmetic and type punning
target_compile_options(hail-to-the-king PRIVATE -fwrapv -fno-strict-aliasing -ffp-contract=off -Wno-unused-variable -Wno-unused-label -Wno-tautological-compare)
if(NOT MSVC)
  target_link_libraries(hail-to-the-king PRIVATE m)
endif()

# PLATFORM=headless (EGL surfaceless, PNG screenshots) or sdl (window + gamepad)
set(PLATFORM "headless" CACHE STRING "headless or sdl")
set(EXT_INC "" CACHE PATH "extra include dir (e.g. the folder holding SDL2/SDL.h)")
target_include_directories(hail-to-the-king PRIVATE ${CMAKE_SOURCE_DIR}/third_party/khronos)
if(EXT_INC)
  target_include_directories(hail-to-the-king PRIVATE ${EXT_INC})
endif()
if(PLATFORM STREQUAL "sdl")
  target_sources(hail-to-the-king PRIVATE runtime/platform_sdl.c)
  set(SDL2_LIB "SDL2" CACHE STRING "SDL2 library to link")
  target_link_libraries(hail-to-the-king PRIVATE ${SDL2_LIB})
else()
  target_sources(hail-to-the-king PRIVATE runtime/platform_headless.c)
  option(NO_EGL "headless without EGL (run with DC_NOGL=1)" OFF)
  if(NO_EGL)
    target_compile_definitions(hail-to-the-king PRIVATE DC_NO_EGL=1)
  else()
    target_link_libraries(hail-to-the-king PRIVATE -l:libEGL.so.1)
  endif()
endif()
if(WIN32)
  target_link_libraries(hail-to-the-king PRIVATE winpthread)
endif()
